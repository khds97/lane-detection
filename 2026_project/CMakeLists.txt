cmake_minimum_required(VERSION 3.16)
project(bev_lane_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(lane_detection_bev lane_detection_bev.cpp)
target_include_directories(lane_detection_bev PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lane_detection_bev PRIVATE ${OpenCV_LIBS})
